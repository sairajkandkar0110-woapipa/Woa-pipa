#!/sbin/bash
. common
disk=/dev/block/sda
max=$(($(blockdev --getsize64 $disk) / 1000000))
[ -z "$max" ] && { echo "Failed to get disk size"; exit 1; }

# Dynamically fetch where userdata starts on Pipa
RES_START=$(parted -s $disk unit MB print | grep -i "userdata" | awk '{print $2}' | tr -d 'MB')
[ -z "$RES_START" ] && { echo "Failed to locate userdata partition start"; exit 1; }

RES_END=7000
winmax=$(( (max - RES_START - RES_END)/1000 ))

[ -n "$1" ] && s="$1" || { echo -n -e "\nEnter Windows size GB (25-$winmax): "; read s; [ -z "$s" ] && echo -e "Please enter a valid size in GB.\n" && exit 1; }
s=$(echo "$s" | tr -cd '0-9.')

case $s in *.*) i=${s%.*}; f=${s#*.}; pow=1; for j in $(seq 1 ${#f}); do pow=$((pow*10)); done
  mb=$((i*1000 + f*1000/pow)) ;; *) mb=$((s*1000)) ;; esac

[ $mb -lt 25000 ] || [ $mb -gt $((winmax*1000)) ] && { echo -n -e "\nSize should be between (25-$winmax)\n"; exit 1; }

# Allocate 350MB for ESP, rest for WIN
esp_end=$(( RES_START + 350 ))
winend=$(( esp_end + mb ))

echo -e "\nWindows: $(( mb/1000 )) GB\nAndroid: $(( (max-winend)/1000 )) GB\n"
echo -n "Does this sound good (y/n)? "; read yn
case "$yn" in [Yy]*) ;; *) echo -e "Rethink things carefully and, eventually, you'll come to a conclusion.\n"; exit 1;; esac

removeparts
sgdisk --resize-table 64 $disk
parted -s $disk \
    mkpart esp fat32 ${RES_START}MB ${esp_end}MB \
    mkpart win ntfs ${esp_end}MB ${winend}MB \
    mkpart userdata ext4 ${winend}MB 100%

# Dynamically flag esp on Pipa
ESP_NUM=$(parted -s $disk print | grep -i "esp" | awk '{print $1}')
[ -n "$ESP_NUM" ] && parted -s $disk set $ESP_NUM esp on

sync; formatwindata; formatwinesp; fixgpt
parted $disk p free
echo "done!"
