#!/sbin/bash
win=/dev/block/by-name/win
esp=/dev/block/by-name/esp

removeparts(){
    # Dynamically locate the partition number for userdata/WOA partitions
    ud_idx=$(parted -s /dev/block/sda print | grep -i "userdata" | awk '{print $1}')
    [ -n "$ud_idx" ] && {
        for i in $(seq $ud_idx 45); do
            parted -s /dev/block/sda rm $i 2>/dev/null
        done
    }
}

formatwinesp(){
    mkfs.fat -F32 -s1 $esp -n ESPPIPA
    mkfs.ntfs -f $win -L WINPIPA
}

formatdata(){
    make_f2fs -f -g android /dev/block/by-name/userdata
    mke2fs -t ext4 -F /dev/block/by-name/metadata
}

formatwindata(){
    make_f2fs -f -g android /dev/block/by-name/userdata
    mke2fs -t ext4 -F /dev/block/by-name/metadata
}

restore(){
    disk=/dev/block/sda
    # Dynamically find the end of the last surviving partition
    LAST_PART_END=$(parted -s $disk unit MB print | grep -v 'Free Space' | grep -E '^[0-9]' | tail -n 1 | awk '{print $3}' | tr -d 'MB')
    removeparts
    parted -s $disk mkpart userdata ext4 ${LAST_PART_END}MB 100%
    sync
    formatdata
}

fixgpt(){
  for d in /dev/block/sd[a-f]; do
    [ -b "$d" ] && printf 'x\nj\n\nk\n\nw\nY\n' | gdisk "$d"
  done
}

[ -e /config/usb_gadget/g1/configs/b.1/mass_storage.0 ] && echo "msc is enabled! aborting..." && echo "reboot and run the script again." && exit 1

if getprop sys.usb.config | grep -q "mtp"; then
    echo " "
    echo "run the script again!"
    sleep 1
    setprop sys.usb.config adb
    rm /config/usb_gadget/g1/configs/b.1/f*
    exit
fi

for i in sdcard data; do umount -lf /$i 2>/dev/null; done
dmsetup remove -f userdata 2>/dev/null
for mnt in $(awk '$1 ~ /^\/dev\/block/ {print $2}' /proc/mounts); do umount -lf "$mnt" 2>/dev/null; done

restoregpt() {
  restore_gpt() {
    gpt=$1 idx=$2 backup="/system/bin/gpt$idx"
    [ -f "$backup" ] || { echo "Backup $backup" not found; return 1; }
    for p in ${gpt}*; do umount "$p" 2>/dev/null; swapoff "$p" 2>/dev/null; done
    (mount | grep -q "$gpt" || grep -q "$gpt" /proc/swaps) && { echo "$gpt busy"; return 1; }
    sgdisk --pretend --clear --load-backup="$backup" "$gpt" || { echo "Dry-run fail $gpt"; return 1; }
    sgdisk --clear --load-backup="$backup" "$gpt" || return 1
    if [ "$gpt" = "/dev/block/sda" ]; then
      UD_NUM=$(parted -s $gpt print | grep -i "userdata" | awk '{print $1}')
      [ -n "$UD_NUM" ] && parted -s "$gpt" resizepart $UD_NUM 100%
      sync; formatdata; parted "$gpt" p free
    fi
    echo "$gpt restored"
  }
  if [ $# -eq 0 ]; then
    i=0
    for g in /dev/block/sd[a-f]; do restore_gpt "$g" $i; i=$((i+1)); done
  else
    case $1 in
      [0-5]) g=/dev/block/sd$(printf \\$(printf '%o' $((97+$1)))); restore_gpt "$g" $1 ;;
      *) echo "Usage: restoregpt [0-5]"; return 1 ;;
    esac
  fi
}

[ "$ignorekrjy" = true ] || {
  bad=0
  for i in 1 2 3 4; do [ -e /dev/block/by-name/bk0$i ] || bad=1; done
  if parted -s /dev/block/sda print | grep -iE 'win|esp'; then bad=1; fi
  [ "$bad" = 1 ] && {
    echo "You have modified partition table! Restoring stock will WIPE ALL DATA."
    echo -n "Continue? (y/n): "; read yn
    case $yn in [Yy]*) restoregpt 0;; *) echo "Aborting."; exit 1;; esac
  }
}
